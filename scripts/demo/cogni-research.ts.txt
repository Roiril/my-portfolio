export type ProjectStatus = 'active' | 'dormant' | 'shipped' | 'shelved';

export type TrajectoryKind =
    | 'note'
    | 'decision'
    | 'deadend'
    | 'question'
    | 'artifact'
    | 'milestone';

export interface TrajectoryEntry {
    id: string;
    projectId: string;
    body: string;
    kind: TrajectoryKind;
    source: 'agent' | 'human';
    createdAt: number;
}

export interface ResearchProject {
    id: string;
    name: string;
    question: string;
    currentState: string;
    nextHand: string;
    status: ProjectStatus;
    entries: TrajectoryEntry[];
    lastTouched: number;
}

export interface ReferenceDoc {
    id: string;
    projectId: string;
    name: string;
    title: string;
    body: string;
}
